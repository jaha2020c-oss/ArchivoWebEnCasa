-- 3. TABLAS DE VENTAS Y DETALLES
CREATE TABLE IF NOT EXISTS ventas (
    id_venta SERIAL PRIMARY KEY,
    fecha_venta TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    nombre_mostrar VARCHAR(100) DEFAULT 'Anónimo',
    total_venta NUMERIC(10,2) DEFAULT 0.00,
    metodo_pago VARCHAR(50) DEFAULT 'Efectivo'
);

CREATE TABLE IF NOT EXISTS detalle_ventas (
    id_detalle SERIAL PRIMARY KEY,
    id_venta INT NOT NULL REFERENCES ventas(id_venta) ON DELETE CASCADE,
    id_platillo INT REFERENCES platillos(id_platillo) ON DELETE SET NULL,
    nombre_platillo VARCHAR(150) NOT NULL,
    cantidad INT NOT NULL CHECK (cantidad > 0),
    precio NUMERIC(10,2) NOT NULL CHECK (precio >= 0),
    subtotal NUMERIC(10,2) NOT NULL CHECK (subtotal >= 0)
);

-- 4. VISTA PARA MÉTRICAS DEL DASHBOARD
CREATE OR REPLACE VIEW vista_dashboard_resumen AS
SELECT 
    COALESCE(SUM(total_venta), 0) AS ingresos_totales,
    COUNT(id_venta) AS total_pedidos,
    COALESCE(AVG(total_venta), 0) AS promedio_ticket
FROM ventas;

 -- PASO 4  -  Tabla de reembolsos (Puntos 2, 3 y 4) --
---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS reembolsos (
    id_reembolso     SERIAL PRIMARY KEY,
    id_venta         INT NOT NULL UNIQUE REFERENCES ventas(id_venta) ON DELETE CASCADE,
    motivo           TEXT NOT NULL,
    estado           VARCHAR(20) NOT NULL DEFAULT 'Pendiente'
                     CHECK (estado IN ('Pendiente', 'Aceptado', 'Rechazado')),
    fecha_solicitud  TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_resolucion TIMESTAMPTZ
);


---------------------------------------------------------------------
 -- PASO 5  -  Tabla de códigos de recuperación (Punto 6) --
---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS password_resets (
    id         SERIAL PRIMARY KEY,
    email      VARCHAR(150) NOT NULL,
    code       VARCHAR(100) NOT NULL,      -- se guarda la huella del código, no el código
    expires_at TIMESTAMPTZ NOT NULL,
    intentos   INT NOT NULL DEFAULT 0,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX IF NOT EXISTS idx_password_resets_email ON password_resets (email);
